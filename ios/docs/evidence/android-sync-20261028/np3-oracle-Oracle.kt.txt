package com.ztransfer.filter
import java.security.MessageDigest
import java.io.File
fun main(args:Array<String>) {
 val methods = PhotoFilterRenderer.javaClass.declaredMethods
 val compile = methods.single { it.name == "compileFilter" }.apply { isAccessible = true }
 val pixel = methods.single { it.name == "filterPixel" }.apply { isAccessible = true }
 var seed=0x12345678
 val colors=IntArray(512) { seed=seed*1664525+1013904223;seed } + intArrayOf(0,-1,0x00123456,0x7f123456,0xff000000.toInt(),0xff808080.toInt())
 val adjustments = listOf(LutAdjustments(),LutAdjustments(contrast=17,saturation=-23,highlights=31,shadows=-19))
 File(args[0]).printWriter().use { output ->
  fun emit(kind:String,id:Int,mode:Int,recipe:PhotoFilterParameters) {
   for(strength in listOf(2,50,80,100)) for(alpha in listOf(false,true)) for(finish in 0..1) {
    val compiled=compile.invoke(PhotoFilterRenderer,PhotoFilterSelection(PhotoFilterPreset("oracle","Oracle",recipe),strength,adjustments[finish]),alpha)
    val hash=MessageDigest.getInstance("SHA-256")
    for(color in colors) {
     val result=pixel.invoke(PhotoFilterRenderer,color,compiled) as Int
     for(shift in listOf(24,16,8,0)) hash.update((result ushr shift).toByte())
    }
    output.println(listOf(kind,id,mode,strength,if(alpha)1 else 0,finish,hash.digest().joinToString(""){"%02x".format(it)}).joinToString("|"))
   }
  }
  for(base in ClassicNp3Style.entries) for(mode in 0..2) {
   val p=when(mode) {
    0 -> ClassicNp3PhotoFilterParameters(base,0,0,0,0,0,0,16,100)
    1 -> ClassicNp3PhotoFilterParameters(base,-3,1,2,-1,if(base.monochrome)3 else 0,if(base.monochrome)3 else 0,20,50)
    else -> ClassicNp3PhotoFilterParameters(base,12,-6,-5,3,4,9,28,20,IntArray(257){4096+it*96})
   }
   emit("C",base.id,mode,p)
  }
  for(case in 0..9) {
   val grade=if(case==0) Np3ColorGrading() else Np3ColorGrading(Np3ColorGradingRegion(45,100,50),Np3ColorGradingRegion(120,-100,-50),Np3ColorGradingRegion(240,100,50),listOf(0,50,100)[(case-1)/3],listOf(-100,0,100)[(case-1)%3])
   val curve=if(case%2==0)null else IntArray(257){minOf(32767,it*128)}
   val p=Np3PhotoFilterParameters(-20,10,-7,13,-9,24,PHOTO_FILTER_COLOR_BAND_CENTERS.mapIndexed{i,c->PhotoFilterColorBand(c,i*17-50,i*12-30,i*9-20)},curve,grade)
   emit("F",case,0,p)
  }
 }
 println("Generated 1648 Android-only pixel hashes")
}
